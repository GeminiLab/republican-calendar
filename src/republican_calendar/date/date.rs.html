15pub type Date = DateWithLeap<DelambreRomme>;

17/// A date in the Republican Calendar, parameterized by a leap year system.
18///
19/// [`Date`] is a type alias for [`DateWithLeap<DelambreRomme>`], which uses the Delambre-Romme
20/// leap year system as the default.
21#[derive(Clone, Copy, PartialEq, Eq, Hash)]
22pub struct DateWithLeap<L: LeapSystem> {
23    year: Year,
24    doy: DayOfYear,
25    _leap: PhantomData<L>,
26}
27
28impl<L: LeapSystem> DateWithLeap<L> {
29    /// Create a new Date from a year and a day of year. Returns [`None`] if the input is invalid.
30    pub fn try_from_yo(year: Year, doy: DayOfYear) -> Option<Self> {
31        if doy.is_legal_for_year::<L>(year) {
32            Some(Self {
33                year,
34                doy,
35                _leap: PhantomData,
36            })
37        } else {
38            None
39        }
40    }
41
42    /// Try to create a new Date from a year, month and day of month. Returns [`None`] if the input
43    /// is invalid.
44    pub fn try_from_ymd(year: Year, month: impl TryInto<Month>, day: Day) -> Option<Self> {
45        Self::try_from_yo(year, DayOfYear::try_from_month_day(month, day)?)
46    }
47
48    /// Get the year of the date.
49    pub fn year(&self) -> Year {
50        self.year
51    }
52
53    /// Get the day of year of the date.
54    pub fn day_of_year(&self) -> DayOfYear {
55        self.doy
56    }
57
58    /// Get the month of the date.
59    pub fn month(&self) -> Month {
60        self.doy.into_month_day().0
61    }
62
63    /// Get the day of the month of the date.
64    pub fn day(&self) -> Day {
65        self.doy.into_month_day().1
66    }
67
68    /// The 0-based index of the day in the epoch of the Republican Calendar.
69    ///
70    /// 1st Vendémiaire, Year I (1792-09-22 Gregorian) is day 0.
71    pub fn epoch_days(&self) -> isize {
72        let days_in_previous_years = L::days_in_years_between(1, self.year);
73        days_in_previous_years + (self.doy as isize - 1)
74    }
75
76    /// Convert the date to a Gregorian date.
77    pub fn to_gregorian(&self) -> NaiveDate {
78        FISRT_DAY + chrono::Duration::days(self.epoch_days() as i64)
79    }
80
81    /// Create a Republican date from a Gregorian date.
82    pub fn from_gregorian(date: NaiveDate) -> Self {
83        let days_since_epoch = (date - FISRT_DAY).num_days() as isize;
84        // div 365 to get a first approximation of the year
85        let mut year = (days_since_epoch / DAYS_IN_COMMON_YEAR as isize + 1) as i32;
86
87        let mut days_before_year = L::days_in_years_between(1, year);
88        let mut days_after_year = days_before_year + L::days_in_year(year) as isize;
89
90        // adjust the year until we find the correct one
91        loop {
92            if days_before_year <= days_since_epoch && days_since_epoch < days_after_year {
93                // we got it
94                let day_of_year = (days_since_epoch - days_before_year + 1) as u32;
95
96                return Self {
97                    year,
98                    doy: DayOfYear::try_from(day_of_year).unwrap(),
99                    _leap: PhantomData,
100                };
101            } else if days_since_epoch < days_before_year {
102                year -= 1;
103                days_after_year = days_before_year;
104                days_before_year -= L::days_in_year(year) as isize;
105            } else {
106                // it seems impossible because we will always overestimate the year at first
107                unreachable!();
108            }
109        }
110    }
111
112    #[cfg(feature = "now")]
113    /// Get today's date in the Republican Calendar.
114    pub fn today() -> Self {
115        let today = chrono::Local::now().date_naive();
116        Self::from_gregorian(today)
117    }
118}

120impl<L: LeapSystem> fmt::Debug for DateWithLeap<L> {
121    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
122        write!(
123            f,
124            "Date<{}> {{ year: {}, month: {}, day: {} }}",
125            L::NAME,
126            self.year,
127            self.month() as u32,
128            self.day()
129        )
130    }
131}

136    }

144    #[test]
145    fn test_new_years_in_century_3() {
146        for rc_year in 201..=300 {
147            let rc_date = Date::try_from_ymd(rc_year, 1, 1).unwrap();
148            let greg_date = rc_date.to_gregorian();
149            assert_eq!(greg_date.month(), 9);
150            assert_eq!(greg_date.day(), 22);
151        }
152
153        for greg_year in 1992..=2091 {
154            let greg_date = NaiveDate::from_ymd_opt(greg_year, 9, 22).unwrap();
155            let rc_date = Date::from_gregorian(greg_date);
156            assert_eq!(rc_date.month(), Month::Vendémiaire);
157            assert_eq!(rc_date.day(), 1);
158        }
159    }
160
161    #[test]
162    fn test_invalid_dates() {
163        // Test invalid dates
164        assert!(Date::try_from_ymd(1, 14, 1).is_none()); // Invalid month
165        assert!(Date::try_from_ymd(1, 1, 31).is_none()); // Invalid day in regular month
166        assert!(Date::try_from_ymd(1, 13, 7).is_none()); // Invalid day in Sans-culottides
167    }
168
169    #[test]
170    fn test_sans_culottides() {
171        // Test Sans-culottides in leap and common years
172        let year1_last = Date::try_from_ymd(1, 13, 5).unwrap();
173        let year2_first = Date::try_from_ymd(2, 1, 1).unwrap();
174
175        assert_eq!(
176            year2_first.to_gregorian() - year1_last.to_gregorian(),
177            chrono::Duration::days(1)
178        );
179
180        let year3_last = Date::try_from_ymd(3, 13, 6).unwrap();
181        let year4_first = Date::try_from_ymd(4, 1, 1).unwrap();
182
183        assert_eq!(
184            year4_first.to_gregorian() - year3_last.to_gregorian(),
185            chrono::Duration::days(1)
186        );
187    }

189    #[test]
190    fn test_season_transitions() {
191        // Test boundary dates for season transitions
192        let autumn_start = Date::try_from_ymd(1, Month::Vendémiaire as u32, 1).unwrap();
193        assert_eq!(
194            autumn_start.to_gregorian(),
195            NaiveDate::from_ymd_opt(1792, 9, 22).unwrap()
196        );

198        let winter_start = Date::try_from_ymd(1, Month::Nivôse as u32, 1).unwrap();
199        assert_eq!(
200            winter_start.to_gregorian(),
201            NaiveDate::from_ymd_opt(1792, 12, 21).unwrap()
202        );
203
204        let spring_start = Date::try_from_ymd(1, Month::Germinal as u32, 1).unwrap();
205        assert_eq!(

208        );
209
210        let summer_start = Date::try_from_ymd(1, Month::Messidor as u32, 1).unwrap();
211        assert_eq!(
