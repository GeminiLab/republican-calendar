1497impl DayOfYear {
1498    /// Create a `DayOfYear` from a month (1-13) and day (1-30 for months 1-12, 1-6 for month 13).
1499    pub fn try_from_month_day(month: impl TryInto<Month>, day: Day) -> Option<Self> {
1500        match month.try_into().ok()? {
1501            Month::SansCulottides => {
1502                if day > 0 && day <= DAYS_IN_SANS_CULOTTIDES_LEAP_YEAR {
1503                    DayOfYear::try_from(DAYS_IN_NORMAL_MONTHS + day).ok()
1504                } else {
1505                    None
1506                }
1507            }
1508            m if day > 0 && day <= DAYS_IN_MONTH => {
1509                DayOfYear::try_from((m as u32 - 1) * DAYS_IN_MONTH + day).ok()
1510            }
1511            _ => None,
1512        }
1513    }
1514
1515    /// Convert a `DayOfYear` into a (month, day) tuple.
1516    pub fn into_month_day(self) -> (Month, Day) {
1517        let day_of_year = self as u32;
1518        if day_of_year > DAYS_IN_NORMAL_MONTHS {
1519            (Month::SansCulottides, day_of_year - DAYS_IN_NORMAL_MONTHS)
1520        } else {
1521            let month = (day_of_year - 1) / DAYS_IN_MONTH + 1;
1522            let day = (day_of_year - 1) % DAYS_IN_MONTH + 1;
1523            (Month::try_from(month).unwrap(), day)
1524        }
1525    }
1526
1527    /// Returns `true` if the day is in the Sans-culottides.
1528    pub fn is_sans_culottides(self) -> bool {
1529        self as u32 > DAYS_IN_NORMAL_MONTHS
1530    }
1531
1532    /// Returns `true` if the day is the leap day (Fête de la Révolution).
1533    pub fn is_leap_day(self) -> bool {
1534        matches!(self, DayOfYear::LaFêteDeLaRévolution)
1535    }
1536
1537    /// Returns `true` if the day is legal for the given year in the given leap year system.
1538    pub fn is_legal_for_year<L: LeapSystem>(self, year: Year) -> bool {
1539        !self.is_leap_day() || L::is_leap_year(year)
1540    }
1541}
